## Explanation
Recipe 6: Pruning and Quantization with Mapping Networks.

This recipe demonstrates how to combine structural compression techniques like
pruning and quantization with mapping networks.

- **Mapping Networks**: Reduce *trainable* parameters and training memory footprint.
- **Pruning & Quantization**: Reduce *inference* storage, parameters, and time.

Because target parameters are generated dynamically, we can prune the fixed
*mapping projection weights* (inside the mappers) to make the weight generation
sparse and extremely fast. We can also quantize the parameters to float16 or
int8 for deployment.

## Key Parts
```python
class SimpleClassificationModel(nn.Module):
    def __init__(self) -> None:
        super().__init__()
        self.fc = nn.Sequential(
            nn.Linear(8, 16),
            nn.ReLU(),
            nn.Linear(16, 2),
        )

    def forward(self, x: torch.Tensor) -> Any:
        return self.fc(x)


def get_sparsity(tensor: torch.Tensor) -> float:
    """Calculate the fraction of zero elements in a tensor."""
    zeros = (tensor == 0).sum().item()
    total = tensor.numel()
    return zeros / total


def main() -> None:
    torch.manual_seed(42)

    # 1. Setup synthetic data
    x_train = torch.randn(128, 8)
    y_train = torch.randint(0, 2, (128,))
    train_loader = DataLoader(TensorDataset(x_train, y_train), batch_size=32, shuffle=True)

    # 2. Build model and train
    target = SimpleClassificationModel()
    model = MappingModel(target, latent_dim=16, strategy="layerwise")

    print("--- Step 1: Training standard Mapping Model ---")
    loss_fn = MappingLoss(task_loss=ClassificationLoss())
    config = TrainerConfig(max_epochs=3, learning_rate=0.01)

    trainer = MappingTrainer(model=model, train_loader=train_loader, loss_fn=loss_fn, config=config)
    trainer.fit()

    print("\n--- Step 2: Applying 50% unstructured pruning to Mapping Mappers ---")
    # In MappingModel, parameters are generated by Mapper modules.
    # Let's inspect the mapper modules inside the generator.
    # Each layer has a mapper that is an instance of BaseMapper (e.g. MLPMapper)
    generator: Any = model.generator

    print("Pre-pruning sparsity of mappers:")
    for layer in generator.layers:
        # layer has a mapper
        mapper = layer.mapper
        # If mapper has linear layers (e.g. projection matrix), we prune it
        for name, module in mapper.named_modules():
            if isinstance(module, nn.Linear):
                sparsity = get_sparsity(module.weight)
                print(f"  {layer.name} ({name}) weight sparsity: {sparsity * 100:.1f}%")

    print("\nPruning 50% of the weights in projection layers...")
    for layer in generator.layers:
        mapper = layer.mapper
        for name, module in mapper.named_modules():
            if isinstance(module, nn.Linear):
                # Apply PyTorch L1 unstructured pruning
                prune.l1_unstructured(module, name="weight", amount=0.5)  # type: ignore[no-untyped-call]
                # Make the pruning permanent
                prune.remove(module, "weight")  # type: ignore[no-untyped-call]

    print("\nPost-pruning sparsity of mappers:")
    for layer in generator.layers:
        mapper = layer.mapper
        for name, module in mapper.named_modules():
            if isinstance(module, nn.Linear):
                sparsity = get_sparsity(module.weight)
                print(f"  {layer.name} ({name}) weight sparsity: {sparsity * 100:.1f}%")
                assert abs(sparsity - 0.5) < 0.05, "Pruning was not applied correctly!"

    print("\n--- Step 3: Quantizing model weights to Float16 for inference ---")
    # We can quantize the entire MappingModel to half precision (float16)
    # to save memory and accelerate inference on edge devices or GPUs.
    model_fp16 = model.half()

    print("Checking datatypes of components:")
    # Check latent vector type
    for name, param in model_fp16.named_parameters():
        if param.requires_grad:
            print(f"  Latent parameter '{name}' dtype: {param.dtype}")
            assert param.dtype == torch.float16, "Latent vector is not FP16!"

    # Run inference with the FP16 model to verify correctness
    inputs = torch.randn(2, 8).half()
    res = model_fp16(inputs)
    print(f"FP16 Output Shape: {res.predictions.shape}, Dtype: {res.predictions.dtype}")
    assert res.predictions.dtype == torch.float16, "Outputs are not FP16!"

    print("\nPruning and quantization successfully applied to the Mapping Network!")
```
